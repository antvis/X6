import { Graph } from '@antv/x6'
import insertCss from 'insert-css'

const container = document.getElementById('container')
const graph = new Graph({
  container,
  grid: true,
})

graph.addEdge({
  source: { x: 240, y: 40 },
  target: { x: 280, y: 180 },
  vertices: [{ x: 240, y: 140 }],
})

graph.addEdge({
  source: { x: 340, y: 40 },
  target: { x: 380, y: 180 },
  vertices: [{ x: 340, y: 140 }],
  connector: {
    name: 'rounded',
    args: { radius: 10 },
  },
  attrs: {
    line: {
      targetMarker: 'classic',
      stroke: '#f5222d',
    },
  },
})

graph.addEdge({
  source: { x: 440, y: 40 },
  target: { x: 480, y: 180 },
  vertices: [{ x: 440, y: 140 }],
  connector: { name: 'smooth' },
  attrs: {
    line: {
      stroke: '#faad14',
      targetMarker: 'classic',
    },
  },
})

graph.addEdge({
  source: { x: 540, y: 40 },
  target: { x: 580, y: 180 },
  vertices: [{ x: 540, y: 140 }],
  connector: { name: 'smooth' },
  attrs: {
    line: {
      stroke: '#1890ff',
      strokeDasharray: 5,
      targetMarker: 'classic',
      style: {
        animation: 'ant-line 30s infinite linear',
      },
    },
  },
})
